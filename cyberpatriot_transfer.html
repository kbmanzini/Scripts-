#!/bin/bash

# CyberPatriot Security Hardening Script
# Run with: sudo bash script.sh
# IMPORTANT: Review and test each section before running in competition

echo "========================================="
echo "CyberPatriot Security Hardening Script"
echo "========================================="
echo ""

# Check if running as root
if [ "$EUID" -ne 0 ]; then 
    echo "Please run as root (use sudo)"
    exit 1
fi

# Create backup directory
BACKUP_DIR="/root/cyberpatriot_backups_$(date +%Y%m%d_%H%M%S)"
mkdir -p "$BACKUP_DIR"
echo "[+] Backup directory created: $BACKUP_DIR"
echo ""

# Function to backup files
backup_file() {
    if [ -f "$1" ]; then
        cp "$1" "$BACKUP_DIR/$(basename $1).bak"
        echo "[BACKUP] $1"
    fi
}

echo "========================================="
echo "1. ROOT PASSWORD SECURITY (4 pts)"
echo "========================================="
echo "[*] Checking root password status..."
getent shadow root
echo "[*] Locking root password..."
passwd -l root
echo "[*] Verifying root password is locked..."
getent shadow root
echo "[✓] Root password locked"
echo ""

echo "========================================="
echo "2. MINIMUM PASSWORD LENGTH (3 pts)"
echo "========================================="
backup_file "/etc/pam.d/common-password"
echo "[*] Setting minimum password length to 10..."
sed -i '/pam_unix.so/ s/$/ minlen=10/' /etc/pam.d/common-password
echo "[✓] Minimum password length configured"
echo ""

echo "========================================="
echo "3. PASSWORD HISTORY (3 pts)"
echo "========================================="
echo "[*] Configuring password history (remember 3)..."
sed -i '/pam_unix.so/ s/$/ remember=3/' /etc/pam.d/common-password
echo "[✓] Password history configured"
echo ""

echo "========================================="
echo "4. ACCOUNT LOCKOUT POLICY (4 pts)"
echo "========================================="
echo "[*] Creating faillock configuration files..."

# Create faillock config
cat > /usr/share/pam-configs/faillock << 'EOF'
Name: Lockout on failed logins
Default: no
